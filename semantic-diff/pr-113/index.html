Claim a literal at a union with exact keys where an arm states an optional field it omits

verified
prepare.ts#measureAnalysis

Why

Pretext's prepare.ts#measureAnalysis returns { widths, ..., segments } at (PreparedLineBreakData & { segments: string[] }) | (PreparedLineBreakData & { segments?: never }), whose arms state an optional items?: never. The producer claimed the literal as an open record, which cannot meet an arm stating a key it may own, so search found no derivation of the return-contract claim. A fresh literal owns exactly its written keys; the producer already claims that bound where a single contract states an optional field, and not where the expected type is a union no written field selects one arm of.

type Data = { widths: number[]; items?: never }

export function measure(keep: boolean): (Data & { segments: string[] }) | (Data & { segments?: never }) {
  const widths: number[] = [1, 2]
  const segments = keep ? ["a", "b"] : null
  if (segments !== null) return { widths, segments }
  return { widths }
}

export function total(keep: boolean): number {
  const data = measure(keep)
  return data.segments === undefined ? data.widths.length : data.widths.length + data.segments.length
}
