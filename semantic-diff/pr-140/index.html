  • stated-region-steps positive new → verified push and a nested call rebased to the region
    // The step-frame rule (FOOTPRINTS.md, "Argument-path store regions"): each
// step's frame, read at its own positions, is rebased to the enclosing
// entry through the slot it reads and included in the stated frame.
// `note`'s `push` changes the store of the array `runs.open` reads (the
// narrower frame `ReceiverGrows` states, every other old location kept);
// `scan` changes it only by calling `note` with its own `runs`, whose
// frame rebases to `scan`'s region through the stable field. `measure`
// allocates the record, so its literal's array is apart from `flags`, and
// keeps `i < flags.length` across each call of `scan`.
type Runs = { /** @stable */ readonly open: number[] };

function note(runs: Runs, c: number): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {
  runs.open.push(c, 0);
}

function scan(runs: Runs, text: string): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {
  for (let i = 0; i < text.length; i++) note(runs, text.charCodeAt(i));
}

export function measure(flags: number[]): number {
  const runs: Runs = { open: [] };
  let sum = 0;
  for (let i = 0; i < flags.length; i++) {
    scan(runs, "x");
    sum += flags[i]!;
  }
  return sum;
}

/** @requires safeInteger(k) */

    if (0 <= k && k < open.length) open[k] = 1;
  }
}

    // The step-frame rule's alias boundary (FOOTPRINTS.md, "Argument-path
// store regions"): `scan` clears the array `runs.open` reads by calling
// `clear`, whose frame rebases to `scan`'s region. `aliased` builds the
// record around `flags` itself, so the region is `flags`'s store: no

type Runs = { /** @stable */ readonly open: number[] };

function clear(runs: Runs): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {
  runs.open.splice(0);
}

function scan(runs: Runs): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {
  clear(runs);
}

export function aliased(flags: number[]): number {
  const runs: Runs = { open: flags };
  let sum = 0;
  for (let i = 0; i < flags.length; i++) {
    scan(runs);
    sum += flags[i]!;
  }
  return sum;
}

    // The step-frame rule's rebasing boundary (FOOTPRINTS.md, "Argument-path
// store regions"): `scan` states the region of the array `runs.open` reads,

function take(runs: Runs): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {

}

function scan(runs: Runs, other: number[]): Modifies<{ values: "any"; "runs.open[*]": "any" }, void> {
  take({ open: other });
}

export function count(xs: number[]): number {
  const runs: Runs = { open: [] };
  let sum = 0;
  for (let i = 0; i < xs.length; i++) {
    scan(runs, xs);
    sum += xs[i]!;
  }
  return sum;
}
