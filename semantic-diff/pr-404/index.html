  • quantified-dense-append-write-gap gap new → gap #395: dense-append boundedWrite does not extend single/parallel every
    // Building two parallel arrays in lockstep via dense-append element writes
// (`if (i > a.length) throw; a[i] = v`) extends both the single-array
// guarded length invariant and the parallel-array length invariant.
// Safe (run under Node).
/**
 * @requires srcAdvances.length === srcExtras.length
 * @requires forall i in srcAdvances: srcAdvances[i] === null || 1 <= srcAdvances[i].length
 * @requires forall i in srcAdvances: srcAdvances[i] === null || srcExtras[i] === null || srcAdvances[i].length <= srcExtras[i].length
 */
export function copyLockstep(
  srcAdvances: (number[] | null)[],
  srcExtras: (number[] | null)[],
): number {
  const advances: (number[] | null)[] = [];
  const extras: (number[] | null)[] = [];
  /**
   * @invariant safeInteger(i) && 0 <= i && i <= srcAdvances.length && advances.length === i && extras.length === i
   * @invariant forall j in advances: advances[j] === null || 1 <= advances[j].length
   * @invariant forall j in advances: advances[j] === null || extras[j] === null || advances[j].length <= extras[j].length
   */
  for (let i = 0; i < srcAdvances.length; i++) {
    const adv = srcAdvances[i] ?? null;
    const ex = srcExtras[i] ?? null;
    if ((i & 0x7fffffff) !== i || i > advances.length) throw new RangeError("index");
    advances[i] = adv;
    if ((i & 0x7fffffff) !== i || i > extras.length) throw new RangeError("index");
    extras[i] = ex;
  }
  let sum = 0;
  for (let i = 0; i < advances.length; i++) {
    const row = advances[i];
    if (row === undefined || row === null) continue;
    const ex = extras[i] ?? null;
    sum += row[0]! + (ex === null ? 0 : ex[0]!);
  }
  return sum;
}
  • quantified-suffix-null-gap gap new → gap #396: prefix-only every cannot express suffix null invariant [i, n)
    // Lazy allocation of a parallel array mid-loop (`prepare.ts:559-564, 633`):
// `extras` is allocated on demand as `new Array(n).fill(null)` and written at
// index `i` only when `ex !== null`. On later iterations where `ex === null`,
// proving `advances[i].length <= extras[i].length` (vacuously, because
// `extras[i] === null`) requires knowing `extras[j] === null` on the unwritten
// suffix `i <= j < n`, which `forall j < stop` cannot express.
// Safe (run under Node).
/**
 * @invariant this.extras === null || this.advances.length <= this.extras.length
 * @invariant forall i in this.advances: this.advances[i] === null || 1 <= this.advances[i].length
 * @invariant this.extras === null || forall i in this.advances: this.advances[i] === null || this.extras[i] === null || this.advances[i].length <= this.extras[i].length
 */
export type LazyParallel = {
  advances: (number[] | null)[];
  extras: (number[] | null)[] | null;
};

/**
 * @requires srcAdvances.length === srcExtras.length
 * @requires forall i in srcAdvances: srcAdvances[i] === null || 1 <= srcAdvances[i].length
 * @requires forall i in srcAdvances: srcAdvances[i] === null || srcExtras[i] === null || srcAdvances[i].length <= srcExtras[i].length
 */
export function buildLazyExtras(
  srcAdvances: (number[] | null)[],
  srcExtras: (number[] | null)[],
): LazyParallel {
  const n = srcAdvances.length;
  const advances: (number[] | null)[] = [];
  let extras: (number[] | null)[] | null = null;
  for (let i = 0; i < n; i++) {
    const adv = srcAdvances[i] ?? null;
    const ex = srcExtras[i] ?? null;
    if (adv !== null && ex !== null) {
      extras ??= new Array<number[] | null>(n).fill(null);
      const store = extras;
      if ((i & 0x7fffffff) !== i || i >= store.length) throw new RangeError("index");
      store[i] = ex;
    }
    if ((i & 0x7fffffff) !== i || i > advances.length) throw new RangeError("index");
    advances[i] = adv;
  }
  return { advances, extras };
}
  • quantified-offset-parallel-gap gap new → gap #397: Bound.parallel cannot express shifted advances[base + s] vs subExtras[s]
    // Shifted parallel indexing (`rich-inline.ts:496-578`): a sub-range builder
// appends `subWidths.length` entries onto `advances` starting at `base` while
// building a 0-indexed `subExtras` whose element `subExtras[s]` matches
// `advances[base + s]`. Copying `subExtras[s]` into `extras[base + s]` keeps
// `advances[i].length <= extras[i].length`, relying on a shifted parallel
// invariant (`advances[base + s]` vs `subExtras[s]`) that `Bound.parallel`
// cannot currently express. Safe (run under Node).
/**
 * @invariant this.advances.length <= this.extras.length
 * @invariant forall i in this.advances: this.advances[i] === null || this.extras[i] === null || this.advances[i].length <= this.extras[i].length
 */
export type ConcatParallel = {
  advances: (number[] | null)[];
  extras: (number[] | null)[];
};

export function appendSubRange(prefix: ConcatParallel, subWidths: number[]): ConcatParallel {
  const advances: (number[] | null)[] = prefix.advances.slice();
  const base = advances.length;
  const subExtras: (number[] | null)[] = [];
  for (let s = 0; s < subWidths.length; s++) {
    const w = subWidths[s]!;
    advances.push([w]);
    subExtras.push([w, 0]);
  }
  const extras: (number[] | null)[] = new Array<number[] | null>(advances.length).fill(null);
  for (let i = 0; i < base; i++) {
    if ((i & 0x7fffffff) !== i || i >= extras.length) throw new RangeError("index");
    extras[i] = prefix.extras[i] ?? null;
  }
  for (let s = 0; s < subExtras.length; s++) {
    const at = base + s;
    if ((at & 0x7fffffff) !== at || at >= extras.length) throw new RangeError("index");
    extras[at] = subExtras[s] ?? null;
  }
  return { advances, extras };
}
  • quantified-bitmask-guard-gap gap new → gap #398: BodyClause.guarded cannot express Uint8Array bitmask guards
    // Bitmask-guarded parallel invariant (`rich-inline.ts:461-468`):
// `PRESERVED_SPACE` segments never carry `extras` (`(flags[i] & 0x7) === 2`
// implies `extras[i] === null`), so overwriting `advances[patchAt]` with a
// fresh non-empty array when `(flags[patchAt] & 0x7) === 2` preserves
// `advances[i].length <= extras[i].length`. `BodyClause.guarded` currently
// supports only nullability tags (`=== null` / `=== undefined`), not bitmask
// comparisons over a parallel `Uint8Array`. Safe (run under Node).
const KIND_BITS = 0x7;
const TEXT = 1;
const PRESERVED_SPACE = 2;

/**
 * @invariant this.flags.length === this.advances.length
 * @invariant this.advances.length === this.extras.length
 * @invariant forall i in this.advances: this.advances[i] === null || 1 <= this.advances[i].length
 * @invariant forall i in this.advances: this.advances[i] === null || this.extras[i] === null || this.advances[i].length <= this.extras[i].length
 */
export type TaggedChunks = {
  flags: Uint8Array;
  advances: (number[] | null)[];
  extras: (number[] | null)[];
};

/** @requires safeInteger(patchAt) && 0 <= patchAt && patchAt < kinds.length */
export function buildAndPatchPreservedSpace(kinds: Uint8Array, patchAt: number): TaggedChunks {
  const n = kinds.length;
  const flags = new Uint8Array(n);
  const advances: (number[] | null)[] = [];
  const extras: (number[] | null)[] = [];
  for (let i = 0; i < n; i++) {
    const k = kinds[i]! & KIND_BITS;
    flags[i] = k;
    if (k === TEXT) {
      advances.push([10, 20]);
      extras.push([1, 2]);
    } else {
      advances.push(null);
      extras.push(null);
    }
  }
  if ((flags[patchAt]! & KIND_BITS) === PRESERVED_SPACE) {
    if ((patchAt & 0x7fffffff) !== patchAt || patchAt >= advances.length) {
      throw new RangeError("index");
    }
    advances[patchAt] = [5, 5, 5];
  }
  return { flags, advances, extras };
}

Draft designs for quantified-array extensions (#396, #397, #398)

Why

Verifying Pretext's quantified-array construction hits three expressive-power limits of quantified index facts beyond the five implementation gaps (#391-#395): lazy mid-loop allocation needing a null-suffix invariant (#396, quantified-suffix-null-gap), sub-range copying needing a shifted parallel-array invariant (#397, quantified-offset-parallel-gap), and kind-bit patching needing a bitmask-guarded invariant over a typed array (#398, quantified-bitmask-guard-gap).
