Preserve unrelated facts through fresh host iteration

verified

Why

The broad host iterator step footprint ({value: any, spine: keep}) loses facts about unrelated module cells at the edge into a for-of body, refusing loops over a Set, Map, string, Intl.Segments, or a union of them even when the body reads the cell only as a string.

    // `for...of` over the host's iterables other than arrays: each loop steps
// the iteration intrinsic of its iterable's kind, which TypeScript's type of
// it picks (SOURCE-MODEL.md, "Async and iteration"). A string yields its code
// points, a `Set` its elements, a `Map` a fresh `[key, value]` per entry, and
// `keys`, `values` and `entries` a host iterator of their collection. None of
// these iterators has a `return`, so a loop left early calls nothing.
export function codePoints(s: string): number {
  let n = 0;
  for (const c of s) {
    if (c === " ") continue;
    n = n + c.length;
  }
  return n;
}

    if (x > 10) return x;
  }
  return fallback;
}

export function keysUntil(m: Map<string, number>, stop: string): string {
  let out = "";
  for (const k of m.keys()) {
    if (k === stop) break;
    out = out + k;
  }
  return out;
}

export function values(m: Map<string, boolean>): number {
  let n = 0;
  for (const v of m.values()) if (v) n = n + 1;
  return n;
}

let wordSegmenter: Intl.Segmenter | null = null;

export function getWordSegmenter(): Intl.Segmenter {
  return (wordSegmenter ??= new Intl.Segmenter(undefined, {
    granularity: "word",
  }));
}

export function clearWordSegmenter(): void {
  wordSegmenter = null;
}

export function wordStarts(text: string, offset: number): number[] {
  const result: number[] = [];
  for (const part of getWordSegmenter().segment(text))
    if (part.index > 0) result.push(offset + part.index);
  return result;
}

export function firstWord(text: string): string {
  const segmenter = new Intl.Segmenter("th", {
    localeMatcher: "lookup",
    granularity: "word",
  });
  for (const segment of segmenter.segment(text)) {
    if (segment.isWordLike === true) return segment.segment;
  }
  return "";
}

export function graphemes(text: string): number {
  let n = 0;
  for (const segment of new Intl.Segmenter(undefined, undefined).segment(text))
    n += segment.input.length > 0 ? 1 : 0;
  return n;
}

  • for-of-map-or-array positive verified → verified narrowed module cell kept across Set | array loop
    // A `for...of` over a union of host kinds: a `Map` or an empty array, a
// `Set` or an array. The loop steps the union's iteration, which steps the
// value as its kind's intrinsic does and yields the union of their elements.
// Stands for the loop over a guard's promises in `collect`
// (declaration-graph.mts).
export function promised(
  m: Map<string, Map<string, number>>,
  k: string,
): number {
  let n = 0;
  for (const [name, value] of m.get(k) ?? []) n += name.length + value;
  return n;
}
