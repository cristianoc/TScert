Keep local field-copy read links across loop heads

verified

Why

A local copy of a field before a loop (const xs = h.xs; for (let i = 0; i < h.xs.length; i++) s += xs[i]!;) was refused at xs[i]! with no derivation of its node-typing claim (issue #351). At a loop head, the pass built initial knowledge only from written assertions and index-atom candidates, so the read link xs ← h.xs was dropped on entry to the loop; across calls inside a loop, read links were also not framed by the call's effect.

  • invalid-loop-head-read-link-write negative new → refused writing h.xs drops xs ← h.xs at the loop head (Node: NaN)
    // Writing `h.xs` inside the loop breaks the read link `xs ← h.xs` on the back
// edge, so the link is dropped at the loop head and `xs[i]!` is refused.
// Unsafe under Node: `sumViaWrites({ xs: [1] })` sets `h.xs = [1, 0]` on
// iteration 0 and reads `xs[1]` (`undefined`) on iteration 1.

export function sumViaWrites(h: { xs: number[] }): number {
  const xs = h.xs;
  let s = 0;
  for (let i = 0; i < h.xs.length; i++) {
    h.xs = [...xs, 0];
    s += xs[i]!;
  }
  return s;
}
  • invalid-loop-head-read-link-call negative new → refused values: "any" call drops xs ← h.xs at the loop head (Node: NaN)
    // Calling a function whose effect may change fields (`values: "any"`) breaks
// the read link `xs ← h.xs` on the back edge, so the link is dropped at the
// loop head and `xs[i]!` is refused.
// Unsafe under Node: `sumViaCalls({ xs: [1] })` sets `h.xs = [1, 0]` on
// iteration 0 and reads `xs[1]` (`undefined`) on iteration 1, returning `NaN`.

function replaceWithLonger(h: {
  xs: number[];
}): Modifies<{ values: "any" }, void> {
  if (h.xs.length < 2) h.xs = [...h.xs, 0];
}

export function sumViaCalls(h: { xs: number[] }): number {
  const xs = h.xs;
  let s = 0;
  for (let i = 0; i < h.xs.length; i++) {
    replaceWithLonger(h);
    s += xs[i]!;
  }
  return s;
}
