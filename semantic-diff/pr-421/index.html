Support shifted parallel quantified clauses, shallow-copy slice, and copy-transfer overwrites (#397)

verified

Why

appendSubRange shallow-copies prefix.advances via slice(), appends subWidths.length entries starting at base while building 0-indexed subExtras whose entry subExtras[s] pairs with advances[base + s], and then copies prefix.extras[i] into extras[i] and subExtras[s] into extras[base + s]. This failed with source ensures refusal: a return of 'appendSubRange' does not establish the postcondition because parallel quantified bounds could only index at i (not i + shift), 0-arg slice() did not state shallow-copy element preservation or a fresh region, element-spine separation was not tracked across array growth, in-bounds overwrites did not transfer a copied source element's quantified body into the written parallel array, and dropping a temporary field-read slot did not restate quantified facts onto the surviving object path.

    advances.length,

/**

 * quantified parallel-length invariant against `prefix.extras`.
 */
export function cloneAdvances(prefix: ConcatParallel): ConcatParallel {
  const advances = prefix.advances.slice();
  return { advances, extras: prefix.extras };
}

/**
 * Reading a shifted parallel element at `base + s - from` establishes the
 * paired inner-array length bound.
 *
 * @requires safeInteger(base) && safeInteger(from) && safeInteger(s)
 * @requires 0 <= s && s < subExtras.length
 * @requires 0 <= base + s - from && base + s - from < advances.length
 * @requires forall k in subExtras: advances[base + k - from] === null || subExtras[k] === null || advances[base + k - from].length <= subExtras[k].length
 */
export function sumShiftedPair(
  advances: (number[] | null)[],
  subExtras: (number[] | null)[],
  base: number,
  from: number,
  s: number,
): number {
  const at = base + s - from;
  const adv = advances[at];
  const ext = subExtras[s];
  if (adv === null || adv === undefined || ext === null || ext === undefined)
    return 0;
  let total = 0;
  for (let i = 0; i < adv.length; i++) {
    total += adv[i]! + ext[i]!;
  }
  return total;
}
