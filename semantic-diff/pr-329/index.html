  • builtin-model-push-refused gap new → unsupported the gate: a spine region "this[*]" takes only "any"; @ensures refuses a rest parameter
    // `push` modelled as code, at its contract (ECMA-262, sec-array.prototype.push;
// tools/array-prototype.mts):
//   { name: "push", from: "Array", signature: "push(...items: T[]): number;",
//     effect: frameEffect(receiverExtendFrame, { receiverGrows: true }) }
// On a normal return the receiver's spine changes by `extend` and every other
// location keeps its contents; the receiver's length grows by the count of
// `items`, which it appends. The specification sets each item at the next
// index, then the length; at a length of 2^32 - 1 that write throws a
// RangeError (ArraySetLength), thrown here before it.
/** @ensures this.length === old(this.length) + items.length */
export function push<T>(this: T[], ...items: T[]): Modifies<{ "this[*]": "extend" }, number> {
  /** @invariant safeInteger(k) && 0 <= k && k <= items.length && this.length === old(this.length) + k */
  for (let k = 0; k < items.length; k++) {
    if (this.length >= 4294967295) throw new RangeError("Invalid array length");
    this[this.length] = items[k];
  }
  return this.length;
}
  • builtin-model-fill-refused gap new → unverified the footprint {} with a fresh result is not established at the return
    // `new Array<T>(n).fill(v)` modelled as code, at its contracts (ECMA-262,
// sec-array-len and sec-array.prototype.fill; tools/intrinsics.mts):
//   %ArrayLength%: (n: number) => number, ensures safeInteger(result),
//     0 <= result <= 2^32 - 1, result === n; a RangeError where n is no array length
//   %ArrayFilled%: <T>(n: number, v: T) => T[] (dense),
//     footprintEffect(emptyFootprint, { fresh: true, ensures: result.length === n })
// The construct checks the length (a RangeError where ToUint32(n) is not n),
// then defines each index below it as `v` on an array no code reached before.
/** @ensures safeInteger(n) && result.length === n */
export function filled<T>(n: number, v: T): Modifies<{}, Fresh<T[]>> {
  if (!Number.isSafeInteger(n) || n < 0 || n > 4294967295) throw new RangeError("Invalid array length");
  const out: T[] = [];
  /** @invariant safeInteger(k) && 0 <= k && k <= n && n <= 4294967295 && safeInteger(n) && out.length === k */
  for (let k = 0; k < n; k++) out[k] = v;
  return out;
}

    // `slice()` of a dense receiver modelled as code, at its contract (ECMA-262,
// sec-array.prototype.slice; tools/array-prototype.mts):
//   { name: "slice", from: "ReadonlyArray", signature: "slice(start?: number, end?: number): T[];",
//     stated: { receiver: readonly T[] (dense), params: [], result: T[] (dense),
//       effect: footprintEffect(emptyFootprint, { fresh: true,
//         ensures: result.length === receiver.length }) } }
// It reads the length once, then defines each element at its index of a
// fresh array (ArraySpeciesCreate at the standard species). Of a dense
// receiver every index below the length is present.
/** @ensures result.length === this.length */
export function slice<T>(this: readonly T[]): Modifies<{}, Fresh<T[]>> {
  const len = this.length;
  const out: T[] = [];
  /** @invariant safeInteger(k) && 0 <= k && k <= len && safeInteger(len) && len === this.length && out.length === k */
  for (let k = 0; k < len; k++) out[k] = this[k];
  return out;
}
  • invalid-builtin-model-* gap new → stopped at its positive's refusal

Gap

The contracts are not met as written: the receiver's spine at extend has no written form; an element write to a locally allocated array counts against a stated frame, and an array literal is not certified fresh; dense presence has no written form; at a generic T an element write at the length keeps only k + 1 <= out.length, and an element read written to an array is claimed never.
