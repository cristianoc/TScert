/** A handle: the widths a walker reads. */
export type H = { readonly widths: readonly number[] };

export function prepare(n: number): H {
  const widths: number[] = [];
  for (let i = 0; i < n; i++) widths.push(1);
  return { widths };
}

export function walk(h: H, visit: () => void): number {
  const w = h.widths;
  let sum = 0;
  for (let i = 0; i < w.length; i++) {
    visit();
    sum += w[i] ?? 0;
  }
  return sum;
}

export function first(xs: number[], visit: () => void): number {
  const c = { i: 0 };
  visit();
  return xs[c.i] ?? 0;
}

no verdict reads the frames. The check gives the inputs of the premise's check (ReadOnly.frameCheck_sound) under the interpretation of the description that the earlier checks assume.
