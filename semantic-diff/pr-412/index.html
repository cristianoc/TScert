Enforce distinct array element cells and preserve quantified facts on in-bounds overwrites (#392)

verified

Why

Overwriting an in-bounds element of an ordinary array (chunks[k] = next, chunks[k] = null, or a guarded nullable array) dropped quantified invariants over chunks (forall i in this.chunks: this.chunks[i] === null || 1 <= this.chunks[i].length), failing replaceChunk with source ensures refusal at [777,781): a return of 'replaceChunk' does not establish the postcondition. The array-spine type contract did not record that distinct indices map to distinct element cells, and the checked-index write rule did not preserve quantified facts over the written array or record the written element's identity.

  • quantified-inbounds-overwrite positive gap → verified in-bounds overwrite with non-null array, null, and guarded nullable value preserves quantified element length
    // Overwriting an in-bounds element of an ordinary array (`chunks[k] = next`,
// `chunks[k] = null`, or a guarded nullable array) keeps the quantified
// invariant on all other indices (`i !== k`), because an ordinary JS array's
// element cells are distinct (`listT`), and checks the body clause at `k`.
// Safe (run under Node).
/**
 * @invariant forall i in this.chunks: this.chunks[i] === null || 1 <= this.chunks[i].length
 */
export type ChunkList = {
  chunks: (number[] | null)[];
};

/** @requires safeInteger(k) && 0 <= k && k < list.chunks.length */
export function replaceChunk(list: ChunkList, k: number): ChunkList {
  const next: number[] = [42];
  const chunks = list.chunks;
  if ((k & 0x7fffffff) !== k || k >= chunks.length)
    throw new RangeError("index");
  chunks[k] = next;
  return list;
}
