type Answered<T> = { readonly ok: true; readonly value: T } | { readonly ok: false; readonly refused: string };

export function pick(m: Map<string, Answered<number>>, k: string): number | null {

  const found = m.get(k);

  return found?.ok ? found.value : null;   // refused: missing field value

}

  • A case example: is the smallest source with the cause's refusal; one made from a census names in its header the census functions it stands for. The test suite (make suite) checks its expected outcome on every run.
  • A gap case: is a safe program TScert refuses for want of a rule or a contract. A negative case is a refusal that is right: the program could produce a wrong value.
  • This census: six of the ten heaviest causes have no case: causes 8 and 9, whose refusals the source above and a 3-line spread of a node list reproduce (measured), and four of the gate refusals of collectDeclarations (slide “collectDeclarations”). Writing their cases is the next step.
